01 / 题目、约束与示例

题意与关键约束 · 学习版

给定非空整数数组，返回一个非空连续子数组的最大元素和。允许出现负数。

阅读官方完整原题 ↗
输入：[-2, 3, -1, 2] 期望结果：4

02 / 先把题目读明白

必须选择非空连续子数组；全负数组也要返回其中一个负数。

先确认：什么输入是合法的、最后返回或修改什么。暂时不需要猜算法名称。

03 / 最直接的做法

固定左端点向右累加，枚举所有区间需要O(n²)。换个角度固定右端点，就能复用上一个结尾的结果。

04 / 从直接做法推导改进

先固定右端点：最优子数组要么从当前数重新开始，要么接在上一个最优结尾之后。end 表示必须以当前位置结尾的最大和；ans 才是所有结尾中的最大值。不要用 0 初始化答案，否则全负数组会错误地选空数组。

05 / 定义状态与更新规则

end是必须以当前元素结尾的最大和；ans是不限定结尾的最大和。end更新为当前数或旧end加当前数的较大值。

from typing import List, Optional
from collections import deque, Counter, defaultdict
import heapq
import math


class Solution:
    def maxSubArray(self, nums: List[int]) -> int:
        end = ans = nums[0]  # 非空：从第一个元素开始
        for i in range(1, len(nums)):
            x = nums[i]
            end = max(x, end + x)  # 重开一段，或延长上一段
            ans = max(ans, end)  # 全局答案与当前结尾状态分开
        return ans

from typing import List, Optional
from collections import deque, Counter, defaultdict
import heapq
import math


class Solution:
    def merge(self, intervals: List[List[int]]) -> List[List[int]]:
        ans = []
        for left, right in sorted(intervals):  # 左端点递增
            if not ans or left > ans[-1][1]:
                ans.append([left, right])  # 不重叠，新开一段
            else:
                ans[-1][1] = max(ans[-1][1], right)  # 保留最远右端
        return ans

from typing import List, Optional
from collections import deque, Counter, defaultdict
import heapq
import math


class Solution:
    def rotate(self, nums: List[int], k: int) -> None:
        if not nums:
            return
        k %= len(nums)  # 整圈移动不改变数组
        def reverse(left, right):
            while left < right:
                nums[left], nums[right] = nums[right], nums[left]
                left += 1
                right -= 1
        reverse(0, len(nums)-1)  # 先交换两段位置
        reverse(0, k-1)  # 恢复前段顺序
        reverse(k, len(nums)-1)

class Solution:
    def sortColors(self, nums: List[int]) -> None:
        low = i = 0
        high = len(nums)-1
        while i <= high:
            if nums[i] == 0:
                nums[low], nums[i] = nums[i], nums[low]
                low += 1; i += 1
            elif nums[i] == 2:
                nums[i], nums[high] = nums[high], nums[i]
                high -= 1  # 新换来的数仍属于未知区

                i += 1

from typing import List, Optional
from collections import deque, Counter, defaultdict
import heapq
import math


class Solution:
    def nextPermutation(self, nums: List[int]) -> None:
        i = len(nums)-2
        while i >= 0 and nums[i] >= nums[i+1]: i -= 1
        if i >= 0:
            j = len(nums)-1
            while nums[j] <= nums[i]: j -= 1
            nums[i], nums[j] = nums[j], nums[i]  # 最小幅度增大前缀
        left, right = i+1, len(nums)-1
        while left < right:  # 降序后缀原地变升序
            nums[left], nums[right] = nums[right], nums[left]
            left += 1; right -= 1

from typing import List, Optional
from collections import deque, Counter, defaultdict
import heapq
import math


class Solution:
    def findDuplicate(self, nums: List[int]) -> int:
        slow = fast = 0
        while True:
            slow = nums[slow]
            fast = nums[nums[fast]]  # 每轮沿边走两步
            if slow == fast: break
        slow = 0  # 起点与相遇点同速走向入口
        while slow != fast:
            slow = nums[slow]
            fast = nums[fast]
        return slow
