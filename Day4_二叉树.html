from typing import List, Optional
from collections import deque, Counter, defaultdict
import heapq
import math

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val, self.left, self.right = val, left, right

class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int:
        if root is None: return 0  # 空树没有层
        left = self.maxDepth(root.left)
        right = self.maxDepth(root.right)
        return max(left, right) + 1

# 以下是本地构造与验证；提交力扣时只保留题目要求的类。
def tree(items):
    if not items or items[0] is None: return None
    root = TreeNode(items[0])
    q, i = deque([root]), 1
    while q and i < len(items):
        node = q.popleft()
        for side in ('left','right'):
            if i < len(items) and items[i] is not None:
                child = TreeNode(items[i])
                setattr(node,side,child)
                q.append(child)
            i += 1
    return root

if __name__ == "__main__":
    assert Solution().maxDepth(tree([1,2,3,None,None,None,4]))==3
    assert Solution().maxDepth(None)==0
    print("自测通过")

from collections import deque, Counter, defaultdict
import heapq
import math

    def __init__(self, val=0, left=None, right=None):
        self.val, self.left, self.right = val, left, right

class Solution:
    def invertTree(self, root: Optional[TreeNode]) -> Optional[TreeNode]:
        if root is None: return None
        left = self.invertTree(root.left)  # 保存翻转后的左子树
        right = self.invertTree(root.right)
        root.left, root.right = right, left
        return root

                child = TreeNode(items[i])
                setattr(node,side,child)
                q.append(child)
            i += 1

def level_values(root):
    if not root: return []
    q, result = deque([root]), []
    while q:
        node = q.popleft()
        result.append(node.val if node else None)
        if node: q.extend([node.left,node.right])
    while result and result[-1] is None: result.pop()
    return result

if __name__ == "__main__":
    assert level_values(Solution().invertTree(tree([1,2,3])))==[1,3,2]
    print("自测通过")

03 / 最直接的做法

from typing import List, Optional
from collections import deque, Counter, defaultdict
import heapq
import math

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val, self.left, self.right = val, left, right

class Solution:
    def diameterOfBinaryTree(self, root: Optional[TreeNode]) -> int:
        ans = 0
        def depth(node):
            nonlocal ans
            if not node: return 0
            left, right = depth(node.left), depth(node.right)
            ans = max(ans, left + right)  # 两条支路在这里相接
            return 1 + max(left, right)  # 只能把一条支路交给父亲
        depth(root)
        return ans

# 以下是本地构造与验证；提交力扣时只保留题目要求的类。
def tree(items):
    if not items or items[0] is None: return None
    root = TreeNode(items[0])
    q, i = deque([root]), 1
    while q and i < len(items):
        node = q.popleft()
        for side in ('left','right'):
            if i < len(items) and items[i] is not None:
                child = TreeNode(items[i])
                setattr(node,side,child)
                q.append(child)
            i += 1
    return root

from typing import List, Optional
from collections import deque, Counter, defaultdict
import heapq
import math

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val, self.left, self.right = val, left, right

class Solution:
    def inorderTraversal(self, root: Optional[TreeNode]) -> List[int]:
        ans, stack, cur = [], [], root
        while cur or stack:
            while cur:
                stack.append(cur)  # 先留待访问，再深入左侧
                cur = cur.left
            cur = stack.pop()
            ans.append(cur.val)
            cur = cur.right
        return ans

03 / 最直接的做法

枚举全部路径会重复计算同一支路贡献。像直径一样在每个节点合并两侧，但父节点只能接走一条支路。

04 / 从直接做法推导改进

与543一样分成“交给父亲的一条支路”和“本节点合并两条支路”。负贡献直接舍弃为0；但答案必须包含至少一个节点，因此ans从负无穷开始。每个转弯点候选为node.val+left+right。

gain返回向父节点的一侧最大贡献，负子贡献截为0；ans从负无穷开始，比较node.val+left+right。

from typing import List, Optional
from collections import deque, Counter, defaultdict
import heapq
import math

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val, self.left, self.right = val, left, right

class Solution:
    def maxPathSum(self, root: Optional[TreeNode]) -> int:
        ans = float('-inf')
        def gain(node):
            nonlocal ans
            if not node: return 0
            left = max(0, gain(node.left))
            right = max(0, gain(node.right))
            ans = max(ans, node.val+left+right)
            return node.val + max(left, right)  # 父节点只能接一边
        gain(root)
        return ans

# 以下是本地构造与验证；提交力扣时只保留题目要求的类。
def tree(items):
    if not items or items[0] is None: return None
    root = TreeNode(items[0])
    q, i = deque([root]), 1
    while q and i < len(items):
        node = q.popleft()
        for side in ('left','right'):
            if i < len(items) and items[i] is not None:
                child = TreeNode(items[i])
                setattr(node,side,child)
                q.append(child)
            i += 1
    return root

    assert Solution().maxPathSum(tree([-3]))==-3
    print("自测通过")
